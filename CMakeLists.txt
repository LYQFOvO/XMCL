cmake_minimum_required(VERSION 3.21)
project(XMCL VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.7 REQUIRED COMPONENTS Quick QuickControls2 Svg ShaderTools)
find_package(Qt6 REQUIRED COMPONENTS Network Quick3D)
find_package(LibArchive REQUIRED)
qt_standard_project_setup(REQUIRES 6.7)
add_library(xmcl_state STATIC src/LauncherController.cpp src/LauncherController.h)
target_link_libraries(xmcl_state PUBLIC Qt6::Core)
target_include_directories(xmcl_state PUBLIC src)
file(GLOB SERVICE_FILES CONFIGURE_DEPENDS "src/services/*.cpp" "src/services/*.h")
add_library(xmcl_services STATIC ${SERVICE_FILES})
target_include_directories(xmcl_services PUBLIC src src/services)
target_link_libraries(xmcl_services PUBLIC Qt6::Core Qt6::Gui Qt6::Network Qt6::Quick3D LibArchive::LibArchive)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/bridge/dist/kamucl-bridge-1.0.1.jar")
    set_source_files_properties(bridge/dist/kamucl-bridge-1.0.1.jar PROPERTIES QT_RESOURCE_ALIAS "bridge/kamucl-bridge.jar")
    qt_add_resources(xmcl_services "bridge_mod" PREFIX "/" FILES bridge/dist/kamucl-bridge-1.0.1.jar)
endif()
if(APPLE)
    target_link_libraries(xmcl_services PRIVATE "-framework Security" "-framework CoreFoundation")
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(tst_controller tests/tst_controller.cpp)
    target_link_libraries(tst_controller PRIVATE xmcl_state Qt6::Test)
    add_test(NAME controller COMMAND tst_controller)
    foreach(suite instances accounts download game java bridge packs personalization connectivity maintenance community)
        qt_add_executable(tst_${suite} tests/tst_${suite}.cpp)
        target_link_libraries(tst_${suite} PRIVATE xmcl_services Qt6::Test)
        add_test(NAME ${suite} COMMAND tst_${suite})
    endforeach()
    add_subdirectory(examples/native-plugin)
    set_tests_properties(personalization PROPERTIES ENVIRONMENT "XMCL_TEST_PLUGIN=$<TARGET_FILE:xmcl_echo>")
endif()

set(QML_FILES
    ui/Main.qml ui/Theme.qml ui/Icon.qml ui/GlassPanel.qml ui/ActionButton.qml
    ui/Tag.qml ui/Toggle.qml ui/RoundedImage.qml ui/HomePage.qml
    ui/InstanceCard.qml ui/LibraryPage.qml ui/DownloadsPage.qml ui/SettingsPage.qml)
list(APPEND QML_FILES ui/FeaturePanel.qml ui/SkinPreview.qml ui/GlassComboBox.qml ui/InstallDialog.qml ui/CommunityDialog.qml ui/InstanceSettingsDialog.qml ui/LaunchCheckDialog.qml)
foreach(qml_file IN LISTS QML_FILES)
    get_filename_component(qml_name "${qml_file}" NAME)
    set_source_files_properties("${qml_file}" PROPERTIES QT_RESOURCE_ALIAS "${qml_name}")
endforeach()
set_source_files_properties(ui/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(xmcl_ui STATIC
    URI XMCL VERSION 1.0 RESOURCE_PREFIX "/qt/qml" CLASS_NAME XMCLPlugin
    OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/XMCL"
    QML_FILES ${QML_FILES})
target_link_libraries(xmcl_ui PUBLIC Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
file(GLOB ICON_FILES CONFIGURE_DEPENDS "assets/icons/*.svg")
qt_add_resources(xmcl_ui "artwork" PREFIX "/" BASE "${CMAKE_CURRENT_SOURCE_DIR}"
    FILES assets/forest.png ${ICON_FILES})
qt_add_shaders(xmcl_ui "glass_shaders" PREFIX "/" FILES shaders/glass.frag)
qt_add_executable(XMCL MACOSX_BUNDLE WIN32 src/main.cpp)
target_link_libraries(XMCL PRIVATE xmcl_state xmcl_services xmcl_ui xmcl_uiplugin)
set_target_properties(XMCL PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.xmcl.launcher"
    MACOSX_BUNDLE_BUNDLE_NAME "XMCL"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
if(BUILD_TESTING)
    qt_add_executable(tst_ui tests/tst_ui.cpp)
    target_link_libraries(tst_ui PRIVATE xmcl_state xmcl_services xmcl_ui xmcl_uiplugin Qt6::Test)
    add_test(NAME ui COMMAND tst_ui)
    set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
